{{- $scheme := "http" }}{{- if $ingress.tls.enabled }}{{ $scheme = "https" }}{{ end }}

  Ingress {{ $host }} (class: {{ include "conxdc.ingress.className" (dict "context" $ "item" $ingress) | default "default" }}, tls: {{ $ingress.tls.enabled }})

{{- $endpoint := index $.Values.controlplane.endpoints $endpointName }}
{{- if $endpoint }}
    {{ printf "%-11s" $endpointName }} {{ $scheme }}://{{ $host }}{{ $endpoint.path }}
{{- end }}
{{- end }}
{{- end }}
{{- end }}
{{- if not $controlplaneHasIngress }}

  No control plane ingress is enabled — endpoints are cluster-internal only.
{{- end }}

  Endpoints NOT exposed via any ingress (cluster-internal only):
{{- $controlplaneExposedEndpointNames := list }}
{{- range $ingress := .Values.controlplane.ingresses }}{{- if $ingress.enabled }}{{- range $exposedEndpointName := $ingress.endpoints }}{{ $controlplaneExposedEndpointNames = append $controlplaneExposedEndpointNames $exposedEndpointName }}{{- end }}{{- end }}{{- end }}
{{- range $endpointName, $endpoint := .Values.controlplane.endpoints }}
{{- if not (has $endpointName $controlplaneExposedEndpointNames) }}
    {{ printf "%-11s" $endpointName }} :{{ $endpoint.port }}{{ $endpoint.path }}
{{- end }}
{{- end }}

  Reach a cluster-internal endpoint via port-forward:
{{- if contains "NodePort" .Values.controlplane.service.type }}
    export NODE_PORT=$(kubectl get -n {{ $namespace }} -o jsonpath="{.spec.ports[0].nodePort}" svc {{ $fullName }}-controlplane)
    export NODE_IP=$(kubectl get nodes -n {{ $namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
    echo "http://$NODE_IP:$NODE_PORT"
{{- else if contains "LoadBalancer" .Values.controlplane.service.type }}
    kubectl get -n {{ $namespace }} svc -w {{ $fullName }}-controlplane
{{- else }}
    export POD_NAME=$(kubectl get pod -n {{ $namespace }} \
      -l app.kubernetes.io/name={{ include "conxdc.name" . }}-controlplane -o name | head -1)
    kubectl -n {{ $namespace }} port-forward $POD_NAME {{ .Values.controlplane.endpoints.management.port }}:{{ .Values.controlplane.endpoints.management.port }}
{{- end }}

  ⚠ WARNING: Vault is running in dev mode (in-memory, no persistence).

{{- if and (eq $mode "hashicorp-persistent") $autoInit }}

  ⚠ WARNING: autoInit is NOT production-safe. It uses a single unseal key
             (key-shares=1) and stores the unseal key and root token as a Kubernetes
             secret in this namespace. Anyone able to read that secret controls the Vault.
             For production use KMS auto-unseal and external key management.

  ⚠ WARNING: The Vault pod becomes sealed again after a restart and is NOT
             unsealed automatically unless an unseal CronJob or KMS auto-unseal is in place.

  ⚠ WARNING: The file audit device is not rotated automatically. Watch the
             auditStorage PVC — a full audit volume can block the Vault.
{{- end }}
